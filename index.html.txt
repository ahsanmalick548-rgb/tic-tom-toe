  <script>
    "use strict";

    const $ = id => document.getElementById(id);
    const icon = name => `<svg class="icon" aria-hidden="true"><use href="#i-${name}"></use></svg>`;
    const randomInt = max => Math.floor(Math.random() * max);
    const clamp = (value, min, max) => Math.max(min, Math.min(max, value));
    const storage = {
      read(key, fallback) {
        try {
          const value = localStorage.getItem("playroom." + key);
          return value === null ? fallback : JSON.parse(value);
        } catch (_) { return fallback; }
      },
      write(key, value) {
        try { localStorage.setItem("playroom." + key, JSON.stringify(value)); }
        catch (_) { /* Games also work when browser storage is unavailable. */ }
      }
    };

    class SoundManager {
      constructor(isEnabled) { this.isEnabled = isEnabled; this.context = null; }
      play(type = "click") {
        if (!this.isEnabled()) return;
        try {
          const AudioContextClass = window.AudioContext || window.webkitAudioContext;
          if (!AudioContextClass) return;
          if (!this.context) this.context = new AudioContextClass();
          if (this.context.state === "suspended") this.context.resume().catch(() => {});
          const sequences = {
            click: [460], move: [420, 560], hit: [360], wall: [220],
            dice: [240, 320, 270, 440], capture: [490, 300, 180],
            point: [540, 720], draw: [420, 420], win: [523, 659, 784, 1047]
          };
          const notes = sequences[type] || sequences.click;
          notes.forEach((frequency, index) => {
            const oscillator = this.context.createOscillator();
            const gain = this.context.createGain();
            const start = this.context.currentTime + index * .075;
            const duration = type === "win" ? .19 : .09;
            oscillator.type = type === "hit" ? "triangle" : "sine";
            oscillator.frequency.setValueAtTime(frequency, start);
            gain.gain.setValueAtTime(0, start);
            gain.gain.linearRampToValueAtTime(type === "wall" ? .018 : .045, start + .008);
            gain.gain.exponentialRampToValueAtTime(.001, start + duration);
            oscillator.connect(gain);
            gain.connect(this.context.destination);
            oscillator.start(start);
            oscillator.stop(start + duration + .02);
            oscillator.onended = () => { oscillator.disconnect(); gain.disconnect(); };
          });
        } catch (_) { /* Audio is optional. */ }
      }
    }

    class TicTacToe {
      static lines = [
        [0,1,2], [3,4,5], [6,7,8],
        [0,3,6], [1,4,7], [2,5,8],
        [0,4,8], [2,4,6]
      ];

      constructor(app) {
        this.app = app;
        this.mode = "ai";
        this.timer = null;
        const saved = storage.read("ttt.scores", {});
        this.scores = {};
        ["X", "O", "draw"].forEach(key => {
          this.scores[key] = Number.isSafeInteger(saved?.[key]) && saved[key] >= 0 ? saved[key] : 0;
        });
        this.cells = Array.from({ length: 9 }, (_, index) => {
          const button = document.createElement("button");
          button.type = "button";
          button.className = "ttt-cell";
          button.addEventListener("click", () => this.playCell(index));
          $("ttt-board").append(button);
          return button;
        });
        document.querySelectorAll("[data-ttt-mode]").forEach(button => {
          button.addEventListener("click", () => {
            if (this.mode === button.dataset.tttMode) return;
            this.mode = button.dataset.tttMode;
            this.app.sound.play("click");
            this.reset();
          });
        });
        $("ttt-difficulty").addEventListener("change", () => this.reset());
        $("ttt-again").addEventListener("click", () => { this.app.sound.play("click"); this.reset(); });
      }

      activate() { this.reset(); }
      deactivate() { clearTimeout(this.timer); }
      reset() {
        clearTimeout(this.timer);
        this.board = Array(9).fill(null);
        this.turn = "X";
        this.result = null;
        this.winningLine = null;
        this.thinking = false;
        this.render();
      }

      static evaluate(board) {
        for (const line of TicTacToe.lines) {
          const [a, b, c] = line;
          if (board[a] && board[a] === board[b] && board[a] === board[c]) return { winner: board[a], line };
        }
        return board.every(Boolean) ? { winner: "draw", line: null } : null;
      }

      minimax(board, maximizing, depth) {
        const outcome = TicTacToe.evaluate(board);
        if (outcome) {
          if (outcome.winner === "O") return 10 - depth;
          if (outcome.winner === "X") return depth - 10;
          return 0;
        }
        let best = maximizing ? -Infinity : Infinity;
        for (let i = 0; i < 9; i++) {
          if (board[i]) continue;
          board[i] = maximizing ? "O" : "X";
          const value = this.minimax(board, !maximizing, depth + 1);
          board[i] = null;
          best = maximizing ? Math.max(best, value) : Math.min(best, value);
        }
        return best;
      }

      chooseAIMove() {
        const open = this.board.map((value, index) => value ? -1 : index).filter(index => index >= 0);
        if ($("ttt-difficulty").value === "easy") return open[randomInt(open.length)];
        let best = -Infinity;
        const moves = [];
        for (const index of open) {
          this.board[index] = "O";
          const score = this.minimax(this.board, false, 0);
          this.board[index] = null;
          if (score > best) { best = score; moves.length = 0; moves.push(index); }
          else if (score === best) moves.push(index);
        }
        return moves[randomInt(moves.length)];
      }

      playCell(index) {
        if (this.result || this.thinking || this.board[index] || (this.mode === "ai" && this.turn === "O")) return;
        this.place(index);
        if (!this.result && this.mode === "ai") {
          this.thinking = true;
          this.render();
          this.timer = setTimeout(() => {
            this.thinking = false;
            this.place(this.chooseAIMove());
          }, this.app.settings.animations ? 420 : 100);
        }
      }

      place(index) {
        if (index === undefined || this.board[index] || this.result) return;
        this.board[index] = this.turn;
        const outcome = TicTacToe.evaluate(this.board);
        if (outcome) {
          this.result = outcome.winner;
          this.winningLine = outcome.line;
          this.scores[this.result]++;
          storage.write("ttt.scores", this.scores);
          this.app.sound.play(this.result === "draw" ? "draw" : "win");
        } else {
          this.turn = this.turn === "X" ? "O" : "X";
          this.app.sound.play("move");
        }
        this.render();
      }

      render() {
        this.cells.forEach((button, index) => {
          const mark = this.board[index];
          if (button.dataset.mark !== (mark || "")) {
            button.innerHTML = mark ? icon(mark.toLowerCase()) : "";
            button.dataset.mark = mark || "";
          }
          button.classList.toggle("o", mark === "O");
          button.classList.toggle("winning", Boolean(this.winningLine?.includes(index)));
          button.disabled = Boolean(mark || this.result || this.thinking);
          button.setAttribute("aria-label", `Row ${Math.floor(index / 3) + 1}, column ${index % 3 + 1}: ${mark || "empty"}`);
        });
        document.querySelectorAll("[data-ttt-mode]").forEach(button => {
          button.setAttribute("aria-pressed", String(button.dataset.tttMode === this.mode));
        });
        $("ttt-difficulty-field").hidden = this.mode !== "ai";
        $("ttt-x-label").textContent = this.mode === "ai" ? "You" : "Player X";
        $("ttt-o-label").textContent = this.mode === "ai" ? "The bot" : "Player O";
        $("ttt-score-x").textContent = this.scores.X;
        $("ttt-score-o").textContent = this.scores.O;
        $("ttt-score-draw").textContent = this.scores.draw;
        $("ttt-again").hidden = !this.result;

    function buildLudoBoard(container, activePlayers, preview = false) {
      container.replaceChildren();
      const putCell = (x, y, className, color, text = "") => {
        const cell = document.createElement("div");
        cell.className = className;
        cell.style.gridColumn = String(x + 1);
        cell.style.gridRow = String(y + 1);
        if (color) cell.style.setProperty("--player-color", color);
        cell.textContent = text;
        container.append(cell);
      };
      LUDO_TRACK.forEach(([x,y], index) => {
        const owner = LUDO_PLAYERS.find(player => player.start === index);
        putCell(x, y, "ludo-cell" + (owner ? " safe" : ""), owner?.color, owner ? "★" : "");
      });
      LUDO_PLAYERS.forEach((player, index) => {
        const home = document.createElement("div");
        home.className = "ludo-home" + (activePlayers.includes(index) ? "" : " inactive");
        home.style.setProperty("--player-color", player.color);
        home.style.gridColumn = `${player.home[0] + 1} / span 6`;
        home.style.gridRow = `${player.home[1] + 1} / span 6`;
        container.append(home);
        player.lane.forEach(([x,y]) => putCell(x, y, "ludo-cell lane", player.color));
      });
      const center = document.createElement("div");
      center.className = "ludo-center";
      center.innerHTML = `<svg viewBox="0 0 60 60" aria-hidden="true">
        <path d="M0 0 30 30 0 60Z" fill="var(--primary)"/>
        <path d="M0 0 60 0 30 30Z" fill="var(--accent)"/>
        <path d="M60 0 60 60 30 30Z" fill="var(--secondary)"/>
        <path d="M0 60 60 60 30 30Z" fill="var(--danger)"/>
        </svg><span aria-hidden="true">★</span>`;
      container.append(center);
      const layer = document.createElement("div");
      layer.className = "ludo-tokens";
      container.append(layer);
      if (preview) {
        LUDO_PLAYERS.forEach(player => {
          player.tokens.forEach(([x,y]) => {
            const token = document.createElement("span");
            token.className = "ludo-token";
            token.style.setProperty("--player-color", player.color);
            positionLudoToken(token, x, y);
            layer.append(token);
          });
        });
      }
      return layer;
    }

    class Ludo {
      constructor(app) {
        this.app = app;
        this.mode = "local";
        this.active = false;
        this.timer = null;
        this.interval = null;
        this.pending = null;

        document.querySelectorAll("[data-ludo-mode]").forEach(button => {
          button.addEventListener("click", () => {
            if (this.mode === button.dataset.ludoMode) return;
            this.mode = button.dataset.ludoMode;
            this.app.sound.play("click");
            this.reset();
          });
        });
        $("ludo-count").addEventListener("change", () => this.reset());
        $("ludo-six").addEventListener("change", () => this.reset());
        $("ludo-roll").addEventListener("click", () => this.roll());
        $("ludo-again").addEventListener("click", () => { this.app.sound.play("click"); this.reset(); });
        document.addEventListener("visibilitychange", () => {
          if (!this.active) return;
          if (document.hidden) this.suspend();
          else this.resume();
        });
      }

      activate() {
        this.active = true;
        this.reset();
      }

      deactivate() {
        this.active = false;
        this.clearAsync();
      }

      clearAsync() {
        clearTimeout(this.timer);
        this.timer = null;
        this.pending = null;
        this.stopDiceAnimation();
      }

      reset() {
        this.clearAsync();
        const count = Number($("ludo-count").value);
        this.players = count === 2 ? [0,2] : count === 3 ? [0,1,2] : [0,1,2,3];
        this.needsSix = $("ludo-six").checked;
        // -1 = yard; 0..50 = shared track; 51..55 = private lane; 56 = home.
        this.tokens = Array.from({ length: 4 }, () => [-1,-1]);
        this.turn = 0;
        this.dice = 0;
        this.phase = "roll";
        this.winner = null;
        this.message = this.mode === "ai"
          ? "You’re Violet. Roll the dice to begin!"
          : "Roll the dice. Your first move awaits.";
        this.layer = buildLudoBoard($("ludo-board"), this.players);
        this.render();
      }

      get currentPlayer() { return this.players[this.turn]; }

      isRobot(playerIndex = this.currentPlayer) {
        return this.mode === "ai" && playerIndex !== this.players[0];
      }

      blocked() {
        return !this.active || document.hidden || this.app.modalOpen();
      }

      // One cancellable, resumable step at a time. Robot turns cannot leak into
      // a reset match, run after navigation, or advance behind an open dialog.
      schedule(callback, delay) {
        clearTimeout(this.timer);
        this.timer = null;
        this.pending = { callback, remaining: delay, started: 0 };
        this.resume();
      }

      suspend() {
        if (this.timer !== null) {
          clearTimeout(this.timer);
          this.timer = null;
          if (this.pending) {
            this.pending.remaining = Math.max(0, this.pending.remaining - (performance.now() - this.pending.started));
          }
        }
        this.stopDiceAnimation();
      }

      resume() {
        if (this.blocked()) return;
        if (this.phase === "rolling") this.startDiceAnimation();
        if (!this.pending || this.timer !== null) return;
        const task = this.pending;
        task.started = performance.now();
        this.timer = setTimeout(() => {
          this.timer = null;
          if (this.pending !== task || !this.active) return;
          if (this.blocked()) {
            task.remaining = 0;
            return;
          }
          this.pending = null;
          task.callback();
        }, task.remaining);
      }

      startDiceAnimation() {
        const animate = this.app.settings.animations && !this.blocked();
        $("ludo-dice").classList.toggle("rolling", animate);
        if (animate && this.interval === null) {
          this.interval = setInterval(() => this.paintDice(randomInt(6) + 1), 75);
        }
      }

      stopDiceAnimation() {
        clearInterval(this.interval);
        this.interval = null;
        $("ludo-dice").classList.remove("rolling");
      }

      queueRobotTurn() {
        if (this.active && this.phase === "roll" && this.isRobot()) {
          this.schedule(() => this.roll(true), this.app.settings.animations ? 750 : 350);
        }
      }

      validMoves() {
        if (this.phase !== "move") return [];
        return this.tokens[this.currentPlayer]
          .map((position, index) => position < 56 && (position >= 0 || !this.needsSix || this.dice === 6) ? index : -1)
          .filter(index => index >= 0);
      }

      paintDice(value) {
        const patterns = {
          0: [4], 1: [4], 2: [0,8], 3: [0,4,8],
          4: [0,2,6,8], 5: [0,2,4,6,8], 6: [0,2,3,5,6,8]
        };
        const dice = $("ludo-dice");
        Array.from(dice.children).forEach((pip, index) => {
          pip.style.visibility = patterns[value].includes(index) ? "visible" : "hidden";
          pip.style.opacity = value === 0 ? ".2" : "1";
        });
        dice.setAttribute("aria-label", value ? `Dice result: ${value}` : "Dice not rolled");
      }

      roll(automated = false) {
        if (this.blocked() || this.phase !== "roll" || this.isRobot() !== automated) return;
        this.phase = "rolling";
        this.message = this.isRobot() ? "The robot is rolling the dice…" : "A little luck goes a long way…";
        this.app.sound.play("dice");
        this.render();
        this.schedule(() => {
          this.stopDiceAnimation();
          // Human players and robots use exactly the same fair roll.
          this.dice = randomInt(6) + 1;
          this.phase = "move";
          const valid = this.validMoves();
          if (valid.length) {
            this.message = this.isRobot()
              ? `Rolled a ${this.dice}. The robot is choosing a token…`
              : `Rolled a ${this.dice}. Tap a glowing token.`;
            this.render();
            if (this.isRobot()) {
              this.schedule(() => this.move(this.chooseRobotMove(), true), this.app.settings.animations ? 700 : 350);
            }
          } else {
            this.phase = "waiting";
            this.message = `A ${this.dice}, but no moves. Passing the dice…`;
            this.render();
            this.schedule(() => this.nextTurn(false), 950);
          }
        }, this.app.settings.animations ? 650 : 150);
      }

      // Estimate whether an opponent can capture this square on its next roll.
      dangerAt(playerIndex, position) {
        if (position < 0 || position > 50) return 0;
        const square = (LUDO_PLAYERS[playerIndex].start + position) % 52;
        if (LUDO_SAFE.includes(square)) return 0;
        let danger = 0;
        this.players.forEach(opponent => {
          if (opponent === playerIndex) return;
          this.tokens[opponent].forEach(opponentPosition => {
            if (opponentPosition < 0 || opponentPosition > 50) return;
            const opponentSquare = (LUDO_PLAYERS[opponent].start + opponentPosition) % 52;
            const distance = (square - opponentSquare + 52) % 52;
            if (distance >= 1 && distance <= 6 && opponentPosition + distance <= 50) {
              danger += 7 - distance;
            }
          });
        });
        return danger;
      }

      // Prefer finishing and captures, then safe squares, releasing tokens,
      // escaping threats, and forward progress. No hidden information is used.
      chooseRobotMove() {
        const moves = this.validMoves();
        const playerIndex = this.currentPlayer;
        let chosen = moves[0];
        let bestScore = -Infinity;
        moves.forEach(index => {
          const oldPosition = this.tokens[playerIndex][index];
          const nextPosition = oldPosition === -1 ? 0 : Math.min(56, oldPosition + this.dice);
          let score = nextPosition * .35 + Math.random() * 2;
          if (oldPosition === -1) score += 35;
          if (nextPosition === 56) {
            score += 1000;
          } else if (nextPosition > 50) {
            score += 40;
          } else {
            const square = (LUDO_PLAYERS[playerIndex].start + nextPosition) % 52;
            const safe = LUDO_SAFE.includes(square);
            if (safe) {
              score += 18;
            } else {
              this.players.forEach(opponent => {
                if (opponent === playerIndex) return;
                this.tokens[opponent].forEach(position => {
                  if (position >= 0 && position <= 50 &&
                      (LUDO_PLAYERS[opponent].start + position) % 52 === square) {
                    score += 95;
                  }
                });
              });
            }
            score -= this.dangerAt(playerIndex, nextPosition) * 4;
            if (this.tokens[playerIndex][1 - index] === nextPosition) score -= safe ? 2 : 9;
          }
          score += this.dangerAt(playerIndex, oldPosition) * 2;
          if (score > bestScore) {
            bestScore = score;
            chosen = index;
          }
        });
        return chosen;
      }

      move(index, automated = false) {
        if (this.blocked() || this.isRobot() !== automated || !this.validMoves().includes(index)) return;
        const playerIndex = this.currentPlayer;
        const oldPosition = this.tokens[playerIndex][index];
        const nextPosition = oldPosition === -1 ? 0 : Math.min(56, oldPosition + this.dice);
        this.tokens[playerIndex][index] = nextPosition;
        this.phase = "waiting";
        let captured = false;

        if (nextPosition <= 50) {
          const trackIndex = (LUDO_PLAYERS[playerIndex].start + nextPosition) % 52;
          if (!LUDO_SAFE.includes(trackIndex)) {
            this.players.forEach(opponent => {
              if (opponent === playerIndex) return;
              this.tokens[opponent].forEach((position, tokenIndex) => {
                if (position >= 0 && position <= 50 &&
                    (LUDO_PLAYERS[opponent].start + position) % 52 === trackIndex) {
                  this.tokens[opponent][tokenIndex] = -1;
                  captured = true;
                }
              });
            });
          }
        }

        const finished = this.tokens[playerIndex].filter(position => position === 56).length;
        if (finished === 2) {
          this.phase = "over";
          this.winner = playerIndex;
          this.message = this.mode === "ai"
            ? (this.isRobot() ? "The robot brought both tokens home. Ready for a rematch?" : "Both your tokens made it home. Nicely played!")
            : "Both tokens made it home. A well-earned victory!";
          this.app.sound.play("win");
          this.render();
          return;
        }

        const actor = this.isRobot() ? "The robot" : this.mode === "ai" ? "You" : LUDO_PLAYERS[playerIndex].name;
        this.message = captured ? `${actor} captured a rival! Back to the yard.`
          : nextPosition === 56 ? `${actor} brought a token home!`
          : this.dice === 6 ? `${actor} moved token ${index + 1}. A six earns another turn!`
          : `${actor} moved token ${index + 1}. Passing the dice…`;
        this.app.sound.play(captured ? "capture" : nextPosition === 56 ? "point" : "move");
        this.render();
        const extraTurn = this.dice === 6;
        this.schedule(() => this.nextTurn(extraTurn), this.app.settings.animations ? 900 : 500);
      }

      nextTurn(extraTurn) {
        if (this.phase === "over" || !this.active) return;
        if (!extraTurn) this.turn = (this.turn + 1) % this.players.length;
        this.phase = "roll";
        this.dice = 0;
        this.message = this.isRobot()
          ? (extraTurn ? "A six earns another turn. The robot will roll again." : "The robot is getting ready to roll…")
          : (extraTurn ? "A six earns another turn. Roll again!" : "Roll the dice to make your move.");
        this.render();
        this.queueRobotTurn();
      }

      render() {
        const current = LUDO_PLAYERS[this.currentPlayer];
        const robot = this.isRobot();
        document.querySelectorAll("[data-ludo-mode]").forEach(button => {
          button.setAttribute("aria-pressed", String(button.dataset.ludoMode === this.mode));
        });
        $("ludo-mode-note").textContent = this.mode === "ai"
          ? "You play Violet. AI robots roll and move automatically."
          : "Pass the dice. Everyone shares this device.";
        const robotCount = this.players.length - 1;
        $("ludo-play-mode").textContent = this.mode === "ai"
          ? `You vs. ${robotCount} robot${robotCount === 1 ? "" : "s"}`
          : "Local multiplayer";
        $("ludo-count-label").textContent = this.mode === "ai" ? "Players, including you" : "Players on this device";
        Array.from($("ludo-count").options).forEach(option => {
          const count = Number(option.value);
          option.textContent = this.mode === "ai"
            ? `You + ${count - 1} robot${count === 2 ? "" : "s"}${count === 2 ? " · opposite corners" : ""}`
            : `${count} players${count === 2 ? " · opposite corners" : count === 4 ? " · a full house" : ""}`;
        });

        let turnText = this.mode === "ai"
          ? (robot ? `${current.name} · robot’s turn` : "Your turn · Violet")
          : `${current.name}’s turn`;
        if (this.winner !== null) {
          turnText = this.mode === "ai"
            ? (robot ? `${current.name}’s robot wins!` : "You win!")
            : `${current.name} wins!`;
        }
        $("ludo-turn").textContent = turnText;
        $("ludo-turn").style.color = current.color;
        $("ludo-message").textContent = this.message;
        $("ludo-roll").disabled = this.phase !== "roll" || robot || !this.active;
        $("ludo-roll").hidden = this.phase === "over";
        $("ludo-roll").innerHTML = robot
          ? icon("cpu") + `<span>${this.phase === "rolling" ? "Rolling…" : "Robot’s turn"}</span>`
          : icon("bolt") + "<span>Roll dice</span>";
        $("ludo-again").hidden = this.phase !== "over";
        $("ludo-dice").classList.toggle("rolling", this.phase === "rolling" && this.app.settings.animations && !this.blocked());
        this.paintDice(this.dice);

        const valid = this.validMoves();
        const entries = [];
        this.players.forEach(playerIndex => {
          const player = LUDO_PLAYERS[playerIndex];
          this.tokens[playerIndex].forEach((position, tokenIndex) => {
            if (position === 56) return;
            let coordinates;
            if (position === -1) coordinates = player.tokens[tokenIndex];
            else if (position <= 50) coordinates = LUDO_TRACK[(player.start + position) % 52];
            else coordinates = player.lane[position - 51];
            entries.push({ playerIndex, tokenIndex, position, x: coordinates[0], y: coordinates[1] });
          });
        });

        this.layer.replaceChildren();
        const groups = new Map();
        entries.forEach(entry => {
          const key = `${entry.x},${entry.y}`;
          if (!groups.has(key)) groups.set(key, []);
          groups.get(key).push(entry);
        });
        groups.forEach(group => {
          group.forEach((entry, offset) => {
            const player = LUDO_PLAYERS[entry.playerIndex];
            const eligible = entry.playerIndex === this.currentPlayer && valid.includes(entry.tokenIndex);
            const movable = eligible && !robot;
            const token = document.createElement("button");
            token.type = "button";
            token.className = "ludo-token" + (movable ? " movable" : eligible && robot ? " robot-ready" : "");
            token.style.setProperty("--player-color", player.color);
            token.textContent = String(entry.tokenIndex + 1);
            token.disabled = !movable;
            token.setAttribute("aria-label",
              `${player.name}${this.isRobot(entry.playerIndex) ? " robot" : ""} token ${entry.tokenIndex + 1}, ${entry.position === -1 ? "in starting yard" : entry.position > 50 ? "in home lane" : "on track"}${movable ? ", available to move" : ""}`);
            let dx = 0, dy = 0;
            if (group.length > 1) {
              const angle = offset / group.length * Math.PI * 2;
              dx = Math.cos(angle) * .19;
              dy = Math.sin(angle) * .19;
            }
            positionLudoToken(token, entry.x + dx, entry.y + dy);
            token.addEventListener("click", () => this.move(entry.tokenIndex));
            this.layer.append(token);
          });
        });

        $("ludo-players").innerHTML = this.players.map(playerIndex => {
          const player = LUDO_PLAYERS[playerIndex];
          const finished = this.tokens[playerIndex].filter(position => position === 56).length;
          const role = this.mode === "ai"
            ? `<span class="player-role">${this.isRobot(playerIndex) ? icon("cpu") + "Robot" : "You"}</span>`
            : "";
          return `<div class="player-row ${playerIndex === this.currentPlayer ? "active" : ""}" style="--player-color:${player.color}">
            <span class="player-color"></span>
            <div class="player-details"><strong><span>${player.name}${playerIndex === this.winner ? " · winner" : ""}</span>${role}</strong><small>${finished} of 2 tokens finished</small></div>
            <span class="finished-dots" aria-hidden="true"><i class="${finished > 0 ? "filled" : ""}"></i><i class="${finished > 1 ? "filled" : ""}"></i></span>
          </div>`;
        }).join("");
      }
    }

    class Pong {
      constructor(app) {
        this.app = app;
        this.canvas = $("pong-canvas");
        this.ctx = this.canvas.getContext("2d");
        this.width = 960;
        this.height = 540;
        this.paddleWidth = 14;
        this.paddleHeight = 98;
        this.keys = new Set();
        this.pointers = new Map();
        this.active = false;
        this.frame = null;
        this.state = "ready";
        const theme = getComputedStyle(document.documentElement);
        this.colors = Object.fromEntries(["bg","surface","surface-2","text","muted","primary","accent","border"].map(name => [name, theme.getPropertyValue("--" + name).trim()]));

        ["pong-mode", "pong-difficulty", "pong-target"].forEach(id => {
          $(id).addEventListener("change", () => this.reset());
        });
        $("pong-start").addEventListener("click", () => this.startOrResume());
        $("pong-pause").addEventListener("click", () => this.togglePause());
        window.addEventListener("keydown", event => this.keyDown(event));
        window.addEventListener("keyup", event => this.keys.delete(event.key.toLowerCase()));
        window.addEventListener("blur", () => { this.keys.clear(); this.pause(); });
        document.addEventListener("visibilitychange", () => {
          if (document.hidden) { this.keys.clear(); this.pause(); }
        });
        this.canvas.addEventListener("pointerdown", event => {
          if (!this.active || this.state !== "playing") return;
          event.preventDefault();
          const box = this.canvas.getBoundingClientRect();
          const side = this.mode === "local" && event.clientX - box.left > box.width / 2 ? "right" : "left";
          this.pointers.set(event.pointerId, side);
          this.canvas.setPointerCapture(event.pointerId);
          this.movePointer(event);
        });
        this.canvas.addEventListener("pointermove", event => this.movePointer(event));
        ["pointerup", "pointercancel", "lostpointercapture"].forEach(name => {
          this.canvas.addEventListener(name, event => this.pointers.delete(event.pointerId));
        });
        if ("ResizeObserver" in window) {
          this.resizeObserver = new ResizeObserver(() => this.resize());
          this.resizeObserver.observe(this.canvas);
        } else {
          window.addEventListener("resize", () => this.resize());
        }
      }

      activate() {
        this.active = true;
        this.reset();
        this.resize();
        this.lastTime = 0;
        cancelAnimationFrame(this.frame);
        this.frame = requestAnimationFrame(time => this.tick(time));
      }

      deactivate() {
        this.active = false;
        this.keys.clear();
        this.pointers.clear();
        cancelAnimationFrame(this.frame);
        if (this.state === "playing") this.state = "paused";
      }

      reset() {
        this.keys.clear();
        this.pointers.clear();
        this.mode = $("pong-mode").value;
        this.difficulty = $("pong-difficulty").value;
        this.target = Number($("pong-target").value);
        $("pong-difficulty").disabled = this.mode === "local";
        this.left = { x: 29, y: (this.height - this.paddleHeight) / 2, score: 0 };
        this.right = { x: this.width - 29 - this.paddleWidth, y: (this.height - this.paddleHeight) / 2, score: 0 };
        this.state = "ready";
        this.aiClock = 0;
        this.aiTarget = this.height / 2;
        this.serveBall(Math.random() < .5 ? -1 : 1);
        $("pong-point").textContent = "Drag on the court to play on touch screens.";
        this.updateUI();
        this.draw();
      }

      serveBall(direction) {
        const angle = (Math.random() - .5) * .8;
        this.ball = {
          x: this.width / 2, y: this.height / 2, radius: 8, speed: 365,
          vx: direction * 365 * Math.cos(angle), vy: 365 * Math.sin(angle)
        };
        this.trail = [];
        this.serveTime = 1;
      }

      startOrResume() {
        if (!this.active || this.app.modalOpen()) return;
        if (this.state === "over") this.reset();
        if (this.state !== "ready" && this.state !== "paused") return;
        this.state = "playing";
        this.keys.clear();
        this.app.sound.play("click");
        $("pong-point").textContent = `First to ${this.target}. Make every rally count.`;
        this.updateUI();
        this.canvas.focus({ preventScroll: true });
      }

      pause() {
        if (!this.active || this.state !== "playing") return;
        this.state = "paused";
        this.keys.clear();
        this.pointers.clear();
        this.updateUI();
      }

      togglePause() {
        if (this.state === "playing") { this.pause(); this.app.sound.play("click"); }
        else if (this.state === "paused") this.startOrResume();
      }

      keyDown(event) {
        if (!this.active || this.app.modalOpen()) return;
        const tag = event.target.tagName;
        if (["INPUT","SELECT","TEXTAREA"].includes(tag)) return;
        const key = event.key.toLowerCase();
        if (["w","s","arrowup","arrowdown"].includes(key)) {
          event.preventDefault();
          this.keys.add(key);
        } else if (key === " " && tag !== "BUTTON") {
          event.preventDefault();
          if (!event.repeat) this.togglePause();
        }
      }

      movePointer(event) {
        const side = this.pointers.get(event.pointerId);
        if (!side || this.state !== "playing") return;
        event.preventDefault();
        const box = this.canvas.getBoundingClientRect();
        const y = (event.clientY - box.top) / box.height * this.height;
        this[side].y = clamp(y - this.paddleHeight / 2, 8, this.height - this.paddleHeight - 8);
      }

      resize() {
        if (!this.active || !this.ball) return;
        const box = this.canvas.getBoundingClientRect();
        if (!box.width || !box.height) return;
        const dpr = Math.min(window.devicePixelRatio || 1, 2);
        this.canvas.width = Math.round(box.width * dpr);
        this.canvas.height = Math.round(box.height * dpr);
        this.draw();
      }

      tick(time) {
        if (!this.active) return;
        const dt = this.lastTime ? Math.min((time - this.lastTime) / 1000, .033) : 0;
        this.lastTime = time;
        if (this.state === "playing") this.update(dt);
        this.draw();
        this.frame = requestAnimationFrame(nextTime => this.tick(nextTime));
      }

      update(dt) {
        const playerSpeed = 490;
        const upLeft = this.keys.has("w") || (this.mode === "ai" && this.keys.has("arrowup"));
        const downLeft = this.keys.has("s") || (this.mode === "ai" && this.keys.has("arrowdown"));
        this.left.y += (Number(downLeft) - Number(upLeft)) * playerSpeed * dt;

        if (this.mode === "local") {
          this.right.y += (Number(this.keys.has("arrowdown")) - Number(this.keys.has("arrowup"))) * playerSpeed * dt;
        } else {
          const profiles = {
            easy: { speed: 215, reaction: .25, error: 55 },
            normal: { speed: 325, reaction: .13, error: 25 },
            hard: { speed: 440, reaction: .065, error: 9 }
          };
          const ai = profiles[this.difficulty];
          this.aiClock -= dt;
          if (this.aiClock <= 0) {
            this.aiClock = ai.reaction;
            let target = this.height / 2;
            if (this.ball.vx > 0) {
              target = this.ball.y;
              if (this.difficulty !== "easy") {
                const travel = Math.max(0, (this.right.x - this.ball.x) / this.ball.vx);
                let projected = this.ball.y + this.ball.vy * travel;
                const span = this.height - 2 * this.ball.radius;
                projected = ((projected - this.ball.radius) % (span * 2) + span * 2) % (span * 2);
                target = this.ball.radius + (projected > span ? span * 2 - projected : projected);
              }
            }
            this.aiTarget = target + (Math.random() - .5) * ai.error * 2;
          }
          const distance = this.aiTarget - (this.right.y + this.paddleHeight / 2);
          this.right.y += clamp(distance, -ai.speed * dt, ai.speed * dt);
        }
        this.left.y = clamp(this.left.y, 8, this.height - this.paddleHeight - 8);
        this.right.y = clamp(this.right.y, 8, this.height - this.paddleHeight - 8);
        if (this.serveTime > 0) {
          this.serveTime -= dt;
          return;
        }

        if (this.app.settings.animations) {
          this.trail.push({ x: this.ball.x, y: this.ball.y });
          if (this.trail.length > 9) this.trail.shift();
        } else this.trail = [];

        const steps = Math.max(1, Math.ceil(this.ball.speed * dt / 7));
        for (let step = 0; step < steps; step++) {
          if (this.stepBall(dt / steps)) break;
        }
      }

      stepBall(dt) {
        const ball = this.ball;
        ball.x += ball.vx * dt;
        ball.y += ball.vy * dt;
        if (ball.y - ball.radius < 0) {
          ball.y = ball.radius;
          ball.vy = Math.abs(ball.vy);
          this.app.sound.play("wall");
        } else if (ball.y + ball.radius > this.height) {
          ball.y = this.height - ball.radius;
          ball.vy = -Math.abs(ball.vy);
          this.app.sound.play("wall");
        }
        for (const [paddle, direction] of [[this.left,1],[this.right,-1]]) {
          const approaching = direction === 1 ? ball.vx < 0 : ball.vx > 0;
          if (approaching &&
              ball.x + ball.radius >= paddle.x &&
              ball.x - ball.radius <= paddle.x + this.paddleWidth &&
              ball.y + ball.radius >= paddle.y &&
              ball.y - ball.radius <= paddle.y + this.paddleHeight) {
            const relative = clamp((ball.y - paddle.y - this.paddleHeight / 2) / (this.paddleHeight / 2), -1, 1);
            const angle = relative * 1.02;
            ball.speed = Math.min(840, ball.speed + 25);
            ball.vx = direction * ball.speed * Math.cos(angle);
            ball.vy = ball.speed * Math.sin(angle);
            ball.x = direction === 1 ? paddle.x + this.paddleWidth + ball.radius + .1 : paddle.x - ball.radius - .1;
            this.app.sound.play("hit");
          }
        }
        if (ball.x < -ball.radius * 2) { this.scorePoint("right"); return true; }
        if (ball.x > this.width + ball.radius * 2) { this.scorePoint("left"); return true; }
        return false;
      }

      scorePoint(side) {
        this[side].score++;
        const name = side === "left" ? (this.mode === "ai" ? "you" : "Player 1") : (this.mode === "ai" ? "the bot" : "Player 2");
        $("pong-point").textContent = `Point to ${name}. ${this.left.score} – ${this.right.score}`;
        this.serveBall(side === "left" ? 1 : -1);
        if (this[side].score >= this.target) {
          this.state = "over";
          this.keys.clear();
          this.pointers.clear();
          this.app.sound.play("win");
        } else this.app.sound.play("point");
        this.updateUI();
      }

      updateUI() {
        $("pong-left-score").textContent = this.left.score;
        $("pong-right-score").textContent = this.right.score;
        $("pong-left-name").textContent = this.mode === "ai" ? "You" : "Player 1";
        $("pong-right-name").textContent = this.mode === "ai" ? "The bot" : "Player 2";
        $("pong-target-label").textContent = `FIRST TO ${this.target}`;
        $("pong-state-label").textContent = ({ ready: "Ready to rally", playing: "Match in play", paused: "Taking a break", over: "Match complete" })[this.state];
        $("pong-overlay").hidden = this.state === "playing";
        $("pong-pause").disabled = !["playing","paused"].includes(this.state);
        $("pong-pause").innerHTML = icon(this.state === "paused" ? "play" : "pause") + `<span>${this.state === "paused" ? "Resume" : "Pause"}</span>`;
        $("pong-controls-note").innerHTML = this.mode === "ai"
          ? `<kbd>W</kbd><kbd>S</kbd><span>or</span><kbd>↑</kbd><kbd>↓</kbd><span>to move · drag to play</span>`
          : `<span>P1</span><kbd>W</kbd><kbd>S</kbd><span>· P2</span><kbd>↑</kbd><kbd>↓</kbd><span>· or drag your half</span>`;

        let title = "Meet your match.";
        let description = "A paddle, a ball, and a little friendly rivalry.<br>Ready to get into the swing of it?";
        let button = "Start match";
        if (this.state === "paused") {
          title = "Take a breather.";
          description = "The rally can wait. Pick up right where you left off.";
          button = "Resume match";
        } else if (this.state === "over") {
          const leftWon = this.left.score >= this.target;
          title = this.mode === "ai"
            ? (leftWon ? "You take the match!" : "The bot wins this one.")
            : `Player ${leftWon ? "1" : "2"} takes the match!`;
          description = `Final score: ${this.left.score} – ${this.right.score}.<br>How about a rematch?`;
          button = "Play again";
        }
        $("pong-overlay-title").textContent = title;
        $("pong-overlay-description").innerHTML = description;
        $("pong-start").innerHTML = icon(this.state === "over" ? "reset" : "play") + `<span>${button}</span>`;
      }

      roundedRect(x, y, width, height, radius) {
        const ctx = this.ctx;
        ctx.beginPath();
        ctx.moveTo(x + radius, y);
        ctx.arcTo(x + width, y, x + width, y + height, radius);
        ctx.arcTo(x + width, y + height, x, y + height, radius);
        ctx.arcTo(x, y + height, x, y, radius);
        ctx.arcTo(x, y, x + width, y, radius);
        ctx.closePath();
        ctx.fill();
      }

      draw() {
        if (!this.ball || !this.canvas.width) return;
        const ctx = this.ctx, colors = this.colors;
        ctx.setTransform(this.canvas.width / this.width, 0, 0, this.canvas.height / this.height, 0, 0);
        ctx.globalAlpha = 1;
        ctx.fillStyle = colors.bg;
        ctx.fillRect(0, 0, this.width, this.height);
        ctx.strokeStyle = colors.muted;
        ctx.lineWidth = 1;
        ctx.globalAlpha = .06;
        ctx.beginPath();
        for (let x = 60; x < this.width; x += 60) { ctx.moveTo(x, 0); ctx.lineTo(x, this.height); }
        for (let y = 60; y < this.height; y += 60) { ctx.moveTo(0, y); ctx.lineTo(this.width, y); }
        ctx.stroke();
        ctx.globalAlpha = .32;
        ctx.lineWidth = 2;
        ctx.setLineDash([9, 13]);
        ctx.beginPath();
        ctx.moveTo(this.width / 2, 18);
        ctx.lineTo(this.width / 2, this.height - 18);
        ctx.stroke();
        ctx.setLineDash([]);
        ctx.globalAlpha = 1;

        for (const [paddle, color] of [[this.left, colors.primary],[this.right, colors.accent]]) {
          ctx.fillStyle = color;
          if (this.app.settings.animations) { ctx.shadowColor = color; ctx.shadowBlur = 13; }
          this.roundedRect(paddle.x, paddle.y, this.paddleWidth, this.paddleHeight, 7);
          ctx.shadowBlur = 0;
        }
        if (this.app.settings.animations) {
          this.trail.forEach((point, index) => {
            ctx.globalAlpha = index / this.trail.length * .22;
            ctx.fillStyle = colors.text;
            ctx.beginPath();
            ctx.arc(point.x, point.y, this.ball.radius * (.3 + index / this.trail.length * .6), 0, Math.PI * 2);
            ctx.fill();
          });
        }
        ctx.globalAlpha = 1;
        ctx.fillStyle = colors.text;
        ctx.beginPath();
        ctx.arc(this.ball.x, this.ball.y, this.ball.radius, 0, Math.PI * 2);
        ctx.fill();
        if (this.state === "playing" && this.serveTime > 0) {
          ctx.fillStyle = colors.muted;
          ctx.font = "600 13px ui-sans-serif, sans-serif";
          ctx.textAlign = "center";
          ctx.fillText(this.serveTime > .45 ? "READY?" : "LET’S RALLY", this.width / 2, this.height / 2 + 45);
        }
      }
    }

    class Playroom {
      constructor() {
        const saved = storage.read("settings", {});
        const prefersReducedMotion = window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
        this.settings = {
          sound: typeof saved?.sound === "boolean" ? saved.sound : true,
          animations: typeof saved?.animations === "boolean" ? saved.animations : !prefersReducedMotion
        };
        this.sound = new SoundManager(() => this.settings.sound);
        this.games = {};
        this.currentGame = null;
        this.toastTimer = null;
        const last = storage.read("lastGame", null);
        this.lastGame = Object.prototype.hasOwnProperty.call(GAME_INFO, last) ? last : null;

        buildLudoBoard($("ludo-preview-board"), [0,1,2,3], true);
        this.applySettings();
        this.updateHome();

        $("brand-home").addEventListener("click", () => this.showHome());
        $("nav-games").addEventListener("click", () => this.showHome());
        $("back-to-menu").addEventListener("click", () => this.showHome());
        $("nav-guide").addEventListener("click", () => this.openRules(null));
        $("game-rules").addEventListener("click", () => this.openRules(this.currentGame));
        $("resume-game").addEventListener("click", () => {
          if (this.lastGame) this.openGame(this.lastGame);
        });
        document.querySelectorAll("[data-game]").forEach(button => {
          button.addEventListener("click", () => this.openGame(button.dataset.game));
        });
        $("reset-game").addEventListener("click", () => {
          if (!this.currentGame) return;
          this.sound.play("click");
          this.games[this.currentGame].reset();
          this.toast("Fresh start. Let’s play.");
        });

        $("sound-toggle").addEventListener("click", () => {
          this.settings.sound = !this.settings.sound;
          this.saveSettings();
          this.sound.play("click");
        });
        $("open-settings").addEventListener("click", () => this.openDialog("settings-dialog"));
        $("setting-sound").addEventListener("change", event => {
          this.settings.sound = event.target.checked;
          this.saveSettings();
          this.sound.play("click");
        });
        $("setting-animations").addEventListener("change", event => {
          this.settings.animations = event.target.checked;
          this.saveSettings();
        });

        document.querySelectorAll("[data-close-dialog]").forEach(button => {
          button.addEventListener("click", () => $(button.dataset.closeDialog).close());
        });
        document.querySelectorAll("dialog").forEach(dialog => {
          dialog.addEventListener("click", event => {
            if (event.target !== dialog) return;
            const rect = dialog.getBoundingClientRect();
            if (event.clientX < rect.left || event.clientX > rect.right ||
                event.clientY < rect.top || event.clientY > rect.bottom) dialog.close();
          });
          dialog.addEventListener("close", () => {
            if (this.currentGame === "ludo") this.games.ludo.resume();
          });
        });
      }

      modalOpen() {
        return Boolean(document.querySelector("dialog[open]"));
      }

      applySettings() {
        document.body.classList.toggle("reduced-motion", !this.settings.animations);
        $("sound-toggle").innerHTML = icon(this.settings.sound ? "sound" : "muted");
        $("sound-toggle").setAttribute("aria-label", this.settings.sound ? "Turn sound off" : "Turn sound on");
        $("sound-toggle").setAttribute("aria-pressed", String(this.settings.sound));
        $("setting-sound").checked = this.settings.sound;
        $("setting-animations").checked = this.settings.animations;
      }

      saveSettings() {
        storage.write("settings", this.settings);
        this.applySettings();
      }

      updateHome() {
        document.querySelectorAll("[data-last]").forEach(badge => {
          badge.hidden = badge.dataset.last !== this.lastGame;
        });
        $("resume-game").hidden = !this.lastGame;
        if (this.lastGame) {
          $("resume-game").querySelector("span").textContent = `Play ${GAME_INFO[this.lastGame].title} again`;
        }
      }

      openGame(id) {
        if (!Object.prototype.hasOwnProperty.call(GAME_INFO, id)) return;
        if (this.currentGame) this.games[this.currentGame].deactivate();
        this.currentGame = id;
        this.lastGame = id;
        storage.write("lastGame", id);
        this.updateHome();
        $("home-view").hidden = true;
        $("game-view").hidden = false;
        document.querySelectorAll(".game-panel").forEach(panel => {
          panel.hidden = panel.id !== id + "-panel";
        });
        $("game-category").textContent = GAME_INFO[id].category;
        $("game-title").textContent = GAME_INFO[id].title;
        if (!this.games[id]) this.games[id] = new GAME_INFO[id].Class(this);
        this.sound.play("click");
        this.games[id].activate();
        window.scrollTo({ top: 0, behavior: "auto" });
        $("game-title").focus({ preventScroll: true });
      }

      showHome() {
        if (this.currentGame) {
          this.games[this.currentGame].deactivate();
          this.sound.play("click");
        }
        this.currentGame = null;
        $("game-view").hidden = true;
        $("home-view").hidden = false;
        document.querySelectorAll(".game-panel").forEach(panel => { panel.hidden = true; });
        this.updateHome();
        window.scrollTo({ top: 0, behavior: "auto" });
      }

      openDialog(id) {
        const dialog = $(id);
        if (dialog.open) return;
        dialog.showModal();
        if (this.currentGame === "pong") this.games.pong.pause();
        if (this.currentGame === "ludo") this.games.ludo.suspend();
      }

      openRules(id) {
        if (id && Object.prototype.hasOwnProperty.call(RULES, id)) {
          $("rules-title").textContent = GAME_INFO[id].title + ": the basics";
          $("rules-content").innerHTML = RULES[id];
        } else {
          $("rules-title").textContent = "The quick guide";
          $("rules-content").innerHTML = Object.keys(GAME_INFO)
            .map(key => `<h3>${GAME_INFO[key].title}</h3>${RULES[key]}`).join("");
        }
        this.openDialog("rules-dialog");
      }

      toast(message) {
        clearTimeout(this.toastTimer);
        $("toast").querySelector("span").textContent = message;
        $("toast").hidden = false;
        this.toastTimer = setTimeout(() => { $("toast").hidden = true; }, 2400);
      }
    }

    new Playroom();
